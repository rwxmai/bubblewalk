cmake_minimum_required(VERSION 3.20)
project(bubblewalk LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(bubble_core analyzer/pe_image.cpp analyzer/interrupted_context.cpp analyzer/provenance.cpp analyzer/demo.cpp)
target_include_directories(bubble_core PUBLIC analyzer)
if(MSVC)
  target_compile_options(bubble_core PRIVATE /W4)
else()
  target_compile_options(bubble_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_executable(bubble-reader-tests tests/reader_tests.cpp)
target_link_libraries(bubble-reader-tests PRIVATE bubble_core)
add_test(NAME bounded_reader COMMAND bubble-reader-tests)
add_executable(bubble-provenance-tests tests/provenance_tests.cpp)
target_link_libraries(bubble-provenance-tests PRIVATE bubble_core)
add_test(NAME frame_provenance COMMAND bubble-provenance-tests)
add_executable(bubble-evidence-tests tests/evidence_tests.cpp)
target_link_libraries(bubble-evidence-tests PRIVATE bubble_core)
add_test(NAME nmi_evidence COMMAND bubble-evidence-tests)
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The analyzer requires an x64 build")
  endif()
  add_executable(bubble analyzer/main.cpp)
  target_link_libraries(bubble PRIVATE bubble_core dbghelp)
  target_compile_definitions(bubble PRIVATE UNICODE _UNICODE)
  if(MSVC)
    target_compile_options(bubble PRIVATE /W4 /EHsc)
  endif()
  add_test(NAME dbghelp_machine_frame COMMAND bubble selftest)
else()
  add_executable(bubble analyzer/portable_main.cpp)
  target_link_libraries(bubble PRIVATE bubble_core)
endif()
add_test(NAME nmi_demo COMMAND bubble demo)
