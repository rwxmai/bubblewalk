cmake_minimum_required(VERSION 3.20)
project(bubblewalk LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(bubble_core analyzer/pe_image.cpp analyzer/interrupted_context.cpp
                        analyzer/provenance.cpp analyzer/demo.cpp analyzer/ioctl_observer.cpp
                        analyzer/ioctl_alert.cpp analyzer/provenance_alert.cpp)
target_include_directories(bubble_core PUBLIC analyzer)
if(MSVC)
  target_compile_options(bubble_core PRIVATE /W4)
else()
  target_compile_options(bubble_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_executable(bubble-reader-tests tests/reader_tests.cpp)
target_link_libraries(bubble-reader-tests PRIVATE bubble_core)
add_test(NAME bounded_reader COMMAND bubble-reader-tests)
add_executable(bubble-provenance-tests tests/provenance_tests.cpp)
target_link_libraries(bubble-provenance-tests PRIVATE bubble_core)
add_test(NAME frame_provenance COMMAND bubble-provenance-tests)
add_executable(bubble-evidence-tests tests/evidence_tests.cpp)
target_link_libraries(bubble-evidence-tests PRIVATE bubble_core)
add_test(NAME nmi_evidence COMMAND bubble-evidence-tests)
add_executable(bubble-provenance-alert-tests tests/provenance_alert_tests.cpp)
target_link_libraries(bubble-provenance-alert-tests PRIVATE bubble_core)
add_test(NAME provenance_alert COMMAND bubble-provenance-alert-tests)
add_executable(bubble-ioctl-tests tests/ioctl_tests.cpp)
target_link_libraries(bubble-ioctl-tests PRIVATE bubble_core)
add_test(NAME ioctl_observer COMMAND bubble-ioctl-tests)
add_executable(bubble-ioctl-alert-tests tests/ioctl_alert_tests.cpp)
target_link_libraries(bubble-ioctl-alert-tests PRIVATE bubble_core)
add_test(NAME ioctl_alert COMMAND bubble-ioctl-alert-tests)
add_executable(bubble-runtime-profile-tests tests/runtime_profile_tests.cpp tests/runtime_profile_c.c)
set_target_properties(bubble-runtime-profile-tests PROPERTIES
                      C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
target_link_libraries(bubble-runtime-profile-tests PRIVATE bubble_core)
add_test(NAME ioctl_runtime_profile COMMAND bubble-runtime-profile-tests)
add_executable(bubble-monitor-timing-tests tests/monitor_timing_tests.cpp)
target_include_directories(bubble-monitor-timing-tests PRIVATE analyzer)
add_test(NAME monitor_timing COMMAND bubble-monitor-timing-tests)
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The analyzer requires an x64 build")
  endif()
  add_executable(bubble analyzer/main.cpp)
  target_link_libraries(bubble PRIVATE bubble_core dbghelp mincore)
  target_compile_definitions(bubble PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  if(MSVC)
    target_compile_options(bubble PRIVATE /W4 /EHsc)
  endif()
  add_test(NAME dbghelp_machine_frame COMMAND bubble selftest)
else()
  add_executable(bubble analyzer/portable_main.cpp)
  target_link_libraries(bubble PRIVATE bubble_core)
endif()
add_test(NAME nmi_replay COMMAND bubble replay)
